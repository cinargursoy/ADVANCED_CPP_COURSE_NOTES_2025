```cpp
#include <iterator>
#include <concepts>
#include <type_traits>

template<typename T>
concept ContainerLike = requires(T a) {

    // Simple Req.
    a.size()
    // Type Req.
    typename T::value_type;
    typename T::iterator;
    typename T::size_type;
    // Compound Req.
    { a.begin() } noexcept -> std::same_as<typename T::iterator>;
    { a.end() } noexcept -> std::same_as<typename T::iterator>;
        // Nested Req.

            { ++it } -> std::same_as<typename T::iterator&>; // pre-increment
            { it != a.end() } -> std::convertible_to<bool>;  // Karsilastirma
```

### *Concept*'lerin *Overloading* Tarafinda Kullanilmasi ve Subsumption

Subsumption nedir:

```cpp
#include <iostream>

template<typename T>
concept has_foo = requires(T x) {
    x.foo();
};

template<typename T>
concept has_foo_bar = requires(T x) {
    x.foo();
    x.bar();
};

void func(has_foo auto)
{
    std::cout << "has_foo auto\n";
}

void func(has_foo_bar auto)
{
    std::cout << "has_foo_bar auto\n";
}

struct A {
    int foo();
};

struct B {
    int foo();
    int bar();
};

int main()
{
    func(A{});
    // func(B{}); // Burada ambiguity olur!
}

/* 
    Out:
    has_foo auto
*/
```

Yukaridaki kodda ambiguity yasaniyor, fakat ayni isi concept ve `&&` operatoru kullanarak subsume edebilirsiniz.

```cpp
#include <iostream>

template<typename T>
concept has_foo = requires(T x) {
    x.foo();
};

template<typename T>
concept has_foo_bar = has_foo<T> && requires(T x) {
    x.bar();
};

void func(has_foo auto)
{
    std::cout << "has_foo auto\n";
}

void func(has_foo_bar auto)
{
    std::cout << "has_foo_bar auto\n";
}

struct A {
    int foo();
};

struct B {
    int foo();
    int bar();
};

int main()
{
    func(A{});
    func(B{});
}

/* 
    Out:
    has_foo auto
    has_foo_bar auto
*/
```

```cpp
#include <concepts>

template<typename T>
concept has_foo = requires(T x) { x.foo(); };

template<typename T>

void func(has_foo auto);
void func(copyable_has_foo auto);
```

# `<ranges>` Kutuphanesi ~ *Since C++20*

STL'in kullanimini buyuk olcude degistiren ama eskisini de koruyan bir ekleme.

Faydalari:
- Ifade gucu STL'e gore daha yuksek.
- Constraint edilmis algoritmalar. Daha acik ve net hata mesajlari.
- Dogrudan container nesnesinin kendisini gonderme avantaji.
- Borrowed range, STL algoritmalarinda tanimsiz davranis olusturabilecek bazi durumlar, sentaks hatasi olusturabiliyor.
- Lazy evaluation.
- Projection

`ranges` bir namespace.

```cpp
    }
```
